function decodeNames(raw) {
  if (!raw || raw.includes('_names')) return [];
  return raw.split(',').map(x => {
    const v = x.trim();
    try { return decodeURIComponent(v); } catch { return v; }
  }).filter(Boolean);
}

main.js: загрузка и применение translate
let translation = null;

async function loadLangData(lang = 'en') {
  const response = await fetch(`translate/${lang}.json`);
  if (!response.ok) throw new Error('translate file not found');
  return response.json();
}

function macroReplace(text, ctx) {
  return String(text)
    .replaceAll('{_offertitle}', ctx.offerTitle || '')
    .replaceAll('{_pricenew}', ctx.pricenew || '')
    .replaceAll('{_priceold}', ctx.priceold || '')
    .replaceAll('{_cc_country}', ctx.cc_country || '')
    .replaceAll('{_name1}', ctx.name1 || 'User');
}

function applyTranslations(dict, ctx) {
  Object.keys(dict).forEach((key) => {
    document.querySelectorAll('.' + key).forEach((el) => {
      el.innerHTML = macroReplace(dict[key], ctx);
    });
  });
}

document.addEventListener('DOMContentLoaded', async () => {
  const names = decodeNames(window.KT.names);
  const ctx = {
    ...window.KT,
    name1: names[0] || 'Alex'
  };
  translation = await loadLangData('en');
  applyTranslations(translation, ctx);
});

Шаг 4. Popup-уведомление (как учебный аналог блока popup_1 из 2934)

Задание: покажи popup через 3-5 секунд, затем скрой. Запомни в localStorage, что уже показывался.

main.js: popup logic
function saveLS(key, value) {
  localStorage.setItem('mini_prefill_' + key, value);
}

function getLS(key) {
  return localStorage.getItem('mini_prefill_' + key);
}

function initPopup() {
  const popup = document.querySelector('.notification');
  const closeBtn = document.querySelector('.notification__closeIcon');
  if (!popup) return;

  if (getLS('popupShowed') === 'true') return;

  setTimeout(() => {
    popup.style.display = 'block';
    saveLS('popupShowed', 'true');
  }, 3500);

  setTimeout(() => {
    popup.style.display = 'none';
  }, 9000);

  closeBtn?.addEventListener('click', () => {
    popup.style.display = 'none';
  });
}

Шаг 5. Валидация формы

Задание: реализуй минимум, как в 2934/4051 — email + phone + required поля.

main.js: validate functions
function validateEmail(email) {
  const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return re.test(String(email).trim().toLowerCase());
}

function validatePhone(phone) {
  return /^\d{5,}$/.test(String(phone).replace(/\D/g, ''));
}

function markField(input, isValid) {
  input.classList.toggle('invalid', !isValid);
  input.classList.toggle('valid', isValid);
}

function validateForm(form) {
  let valid = true;

  const required = form.querySelectorAll('[name][required]');
  required.forEach((el) => {
    const ok = String(el.value || '').trim() !== '';
    markField(el, ok);
    if (!ok) valid = false;
  });

  const email = form.querySelector('.js-email-input');
  if (email) {
    const ok = validateEmail(email.value);
    markField(email, ok);
    if (!ok) valid = false;
  }

  const phone = form.querySelector('.js-phone-input');
  if (phone) {
    const ok = validatePhone(phone.value);
    markField(phone, ok);
    if (!ok) valid = false;
  }

  return valid;
}

Шаг 6. Сбор параметров и отправка (ключевой этап)

Задание: на submit собери все поля в query string и сделай редирект на оффер. Это важно для теста через Display GET Params.

main.js: query-only submit handler
function ensureProductImageParam(form) {
  let hidden = form.querySelector('input[name="sub_id_15"]');
  if (!hidden) {
    hidden = document.createElement('input');
    hidden.type = 'hidden';
    hidden.name = 'sub_id_15';
    form.appendChild(hidden);
  }

  const img = document.querySelector('.def-product-item-image');
  hidden.value = img?.src || window.KT.productImageSrc || '';
}

function buildActionUrl(form) {
  const offerUrl = window.KT.offerUrl;
  if (!offerUrl || offerUrl.includes('{offer}') || offerUrl.includes('%7Boffer%7D')) {
    throw new Error('Offer URL is not resolved');
  }

  const url = new URL(offerUrl, window.location.origin);

  // 1) переносим входящие технические query (в т.ч. _lp/_token)
  const incoming = new URLSearchParams(window.location.search);
  incoming.forEach((value, key) => {
    if (!url.searchParams.has(key) && value !== '') {
      url.searchParams.set(key, value);
    }
  });

  // 2) переносим поля формы
  const fd = new FormData(form);
  for (const [name, value] of fd.entries()) {
    if (typeof value !== 'string') continue;
    if (value.trim() === '') continue;
    url.searchParams.set(name, value.trim());
  }

  // fallback: paymentOption
  if (!url.searchParams.get('paymentOption')) {
    url.searchParams.set('paymentOption', 'card');
  }

  return url.toString();
}

function initForm() {
  const form = document.getElementById('mForm');
  const btn = document.getElementById('startBtn');
  if (!form) return;

  form.addEventListener('submit', (e) => {
    e.preventDefault();

    if (!validateForm(form)) {
      form.querySelector('.invalid')?.focus();
      return;
    }

    ensureProductImageParam(form);

    try {
      btn.disabled = true;
      btn.textContent = (translation?.text_wait || 'Wait') + '...';
      const actionUrl = buildActionUrl(form);
      window.location.href = actionUrl;
    } catch (err) {
      console.error(err);
      btn.disabled = false;
      btn.textContent = 'Continue';
    }
  });
}
Контрольная точка: после submit у тебя в DevTools в Query String Parameters должны быть `sub_id_27, sub_id_24, sub_id_25, sub_id_22, sub_id_9, sub_id_23, sub_id_21, sub_id_26, paymentOption, sub_id_15, _lp, _token`.

Дебаг-сценарий (проверка по шагам)

Сценарий A: локальная проверка

  • Открой страницу с query: ?_lp=1&_token=test123.
  • Заполни форму валидными данными.
  • Проверь, что кнопка меняется на Wait... и идёт редирект.

Сценарий B: через Display GET Params

  • Поставь `window.KT.offerUrl` на тест-оффер отображения GET-параметров.
  • Отправь форму.
  • Убедись, что обязательные поля не пустые, и нет ошибки "не хватает ...".

Сценарий C: ловим типичные баги

Симптом Причина Что чинить
404 после submit {offer} не подставился или URL сломан проверка в buildActionUrl + лог в консоль
Поля пустые в Display GET Params передача ушла в body, а не в query оставить query-only редирект, без form.submit()
Нет `_lp`/`_token` входной URL не содержит их или ты не прокинул входящие query добавь копирование `window.location.search` в action URL
