Раздел 0

Как работать с этим учебником

Это не справочник и не «кусок примера». Это рабочий маршрут разработки. Ты пишешь код в том же порядке, в каком пишет разработчик в реальном проекте.

Маршрут на весь курс

  1. Собрать каркас проекта и comments-UI.
  2. Поднять инфраструктуру: main-script в head, перевод, даты, макросы, общий JS runtime без логики комментариев.
  3. Реализовать legacy-архитектуру comments v1: рабочую, но уязвимую.
  4. Понять на семантическом уровне, почему v1 нестабильна.
  5. Переписать тот же функционал на robust-архитектуру comments v2.
Важно: шаг «legacy v1» здесь не «плохой пример из статьи», а полноценный этап обучения. Ты осознанно собираешь работающую v1, фиксируешь ограничения, и только потом переходишь к v2.

  function handleImageError(img, fileName) {

Checkpoint 2

scripts.js до comments: инфраструктура страницы

На этом этапе comments-движка ещё нет. Мы пишем слой, который подготавливает окружение: переводы, даты, общие утилиты хранения.

Шаг 1. Утилиты хранения и ключи

scripts.js — storage utils
function saveToLocalStorage(param, value) {
  localStorage.setItem(lId + '_' + cId + '_' + param, value);
}

function getFromLocalStorage(param) {
  return localStorage.getItem(lId + '_' + cId + '_' + param);
}

function safeReadJSON(key, fallback) {
  try {
    const raw = localStorage.getItem(key);
    if (!raw) return fallback;
    const parsed = JSON.parse(raw);
    return parsed === null ? fallback : parsed;
  } catch {
    return fallback;
  }
}

function safeWriteJSON(key, value) {
  localStorage.setItem(key, JSON.stringify(value));
}

Семантика шага 1

Что означает lId + '_' + cId + '_' + param

Ты формируешь namespace ключа в localStorage. Это защищает от пересечения данных между разными лендингами и кампаниями.

Почему важно: если не делать namespace, комментарии одного лендинга могут появиться в другом.

const defaultProductName = 'defaultproduct';

function getLangCode() {

async function loadLangData(lang) {
  if (productName === '{sub_id_28}' || !productName) {
    productName = defaultProductName;
  }

  try {
    const response = await fetch(`${productName}/translate/${lang}.json`);
    if (!response.ok) throw new Error(`Language file not found: ${lang}`);
    return await response.json();
  } catch {
    if (lang !== defaultLang) return loadLangData(defaultLang);
    throw new Error('Cannot load translation file');
  }
}

function expandMacros(text) {
  if (typeof text !== 'string') return text;

  const macroMap = {
    '{_product}': () => product,
    '{_supportname}': () => 'UGG',
    '{_name1}': () => 'Alise Djns',
    '{_name2}': () => 'Sam Pit'
  };

  let out = text;
  for (const [token, getter] of Object.entries(macroMap)) {
    out = out.replaceAll(token, getter());
  }
  return out;
}

function formatDateDMY(date) {
  const d = String(date.getDate()).padStart(2, '0');
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const y = date.getFullYear();
  return `${d}.${m}.${y}`;
}

function setDates() {
  const now = new Date();
  const map = {
    '.text_date_DMY': 0,
    '.text_date_DMY_m1': -1,
    '.text_date_DMY_m2': -2
  };

  Object.entries(map).forEach(([selector, offset]) => {
    const date = new Date(now);
    date.setDate(now.getDate() + offset);
    const value = formatDateDMY(date);

    document.querySelectorAll(selector).forEach((el) => {
      el.textContent = value;
    });
  });
}

Семантика шага 2

Почему здесь много функций, а не один большой блок

  • loadLangData отвечает только за сеть и fallback языка.
  • expandMacros отвечает только за замену токенов.
  • setDates отвечает только за time-плейсхолдеры.

Так легче тестировать и переиспользовать каждую часть независимо.

  for (const node of tree) {
    if (node.id === id) return node;

function normalizeComment(raw) {
  const safe = raw && typeof raw === 'object' ? raw : {};
  const likesVal = Number(safe.likes);

  const normalized = {
    id: String(safe.id || `${Date.now()}_${Math.random().toString(16).slice(2)}`),
    text: typeof safe.text === 'string' ? safe.text : '',
    name: typeof safe.name === 'string' && safe.name.trim() ? safe.name.trim() : 'You',
    liked: !!safe.liked,
    likes: Number.isFinite(likesVal) ? Math.max(0, likesVal) : 0,
    date: typeof safe.date === 'string' && safe.date.trim() ? safe.date : formatDateDMY(new Date()),
    replies: [],
    origin: typeof safe.origin === 'string' ? safe.origin : 'user',
    avatar: typeof safe.avatar === 'string' ? safe.avatar : 'user'
  };

  if (Array.isArray(safe.replies)) {
    normalized.replies = safe.replies.map(normalizeComment);
  }

  return normalized;
}

    const initialStaticItems = Array.from(commentsList.querySelectorAll('.comment__item[data-origin="static"]'));
    initialStaticItems.forEach((item, idx) => {
      if (!item.dataset.sid) item.dataset.sid = String(idx);
      item.dataset.threadSid = item.dataset.sid;
      ensureRepliesContainer(item);
      enforceReplyLimit(item);
    });

    initialStaticItems.forEach((item) => {
      const thread = getStaticThread(staticReplyKey, item.dataset.sid);
      if (thread.length) {
        renderStaticRepliesRecursive(item, thread);
      }
    });

    savedComments.forEach((c) => {
      const node = renderSavedComment(c, commentsList);
      if (Array.isArray(c.replies) && c.replies.length) {
        renderUserRepliesRecursive(node, c.replies);
      }
    });

    // post top-level
    textarea.addEventListener('input', () => {
      const valid = textarea.value.trim().length > 0;
      postBtn.disabled = !valid;
      postBtn.classList.toggle('disabled', !valid);
    });

    postBtn.addEventListener('click', () => {
      const text = textarea.value.trim();
      if (!text) return;

      const c = normalizeComment({
        id: `${Date.now()}_${Math.random().toString(16).slice(2)}`,
        name: 'You',
        text,
        likes: 0,
        liked: false,
        date: formatDateDMY(new Date()),
        replies: [],
        origin: 'user',
        avatar: 'user'
      });

      savedComments.push(c);
      safeWriteJSON(storageKey, savedComments);
      renderSavedComment(c, commentsList);

      textarea.value = '';
      postBtn.disabled = true;
      postBtn.classList.add('disabled');
      updateCommentsTotal(block, totalEl);
    });

    // delegated actions
    block.addEventListener('click', (e) => {
      const item = e.target.closest('.comment__item');
      if (!item) return;

      if (e.target.classList.contains('reply-btn')) {
        showInlineReplyFormV1(item, (text) => {
          addReplyRobust(item, text, {
            storageKey,
            staticReplyKey,
            savedComments
          });
          updateCommentsTotal(block, totalEl);
        });
        return;
      }

      if (e.target.classList.contains('like-btn') || e.target.closest('.comment__actions')) {
        toggleLikeRobust(item, storageKey, staticReplyKey, savedComments);
      }
    });

    startOneLiveComment(commentsList, storageKey, savedComments);
    updateCommentsTotal(block, totalEl);
  });
}

document.addEventListener('DOMContentLoaded', initCommentsRobustV2);

  • Лимит глубины предсказуемый на всех ветках (через DOM depth).
  • Один live-комментарий приходит и сохраняется корректно.
  • После reload дерево комментариев восстанавливается без потери 3-й вложенности для корректных веток.

Контроль качества

Чеклист регрессии (прогон после каждой правки)

  1. Стартовая страница открывается без JS-ошибок.
  2. Тексты comments подтягиваются из JSON.
  3. Top-level комментарий публикуется и сохраняется в localStorage.
  4. Reply на static-comment работает.
  5. Reply на user-comment работает.
  6. Reply на reply (3-й уровень) работает и правильно ограничивается.
  7. Лайк/анлайк у static и user-узлов не ломает счётчик.
  8. После reload дерево сохраняется (включая глубину до разрешённого уровня).
  9. Один live-comment приходит по таймеру и не дублируется бесконечно.
  10. В RTL режимe структура и отступы mini-reply корректны.
Практический совет: после каждого крупного шага делай короткий smoke-test по этому чеклисту, чтобы не копить ошибку до конца главы.
